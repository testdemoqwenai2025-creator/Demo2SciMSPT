import pandas as pd
import numpy as np
from sklearn.preprocessing import StandardScaler, LabelEncoder
from sklearn.model_selection import train_test_split

class SyntheticDataPipeline:
    """SciMSPT Synthetic Data Generator"""
    
    def __init__(self, n_samples=5000, random_state=42):
        self.n_samples = n_samples
        self.random_state = random_state
        np.random.seed(random_state)
    
    def generate_gaussian_data(self, means, stds, corr_matrix=None):
        """Generate correlated Gaussian features"""
        n_features = len(means)
        data = np.random.randn(self.n_samples, n_features)
        
        # Apply correlations if provided
        if corr_matrix is not None:
            L = np.linalg.cholesky(corr_matrix)
            data = data @ L.T
        
        # Scale by means and stds
        data = data * stds + means
        
        return pd.DataFrame(data, columns=[f'feature_{i}' for i in range(n_features)])
    
    def normalize(self, df, method='standard'):
        """Apply normalization transformation"""
        if method == 'standard':
            scaler = StandardScaler()
            numeric_cols = df.select_dtypes(include=[np.number]).columns
            df[numeric_cols] = scaler.fit_transform(df[numeric_cols])
        elif method == 'minmax':
            df = (df - df.min()) / (df.max() - df.min())
        return df
    
    def generate_report(self, df):
        """Generate statistical summary report"""
        report = {
            'shape': df.shape,
            'dtypes': df.dtypes.to_dict(),
            'describe': df.describe().to_dict(),
            'missing': df.isnull().sum().to_dict(),
            'correlations': df.corr().to_dict()
        }
        return report


# Execute Pipeline
if __name__ == '__main__':
    pipeline = SyntheticDataPipeline(n_samples=5000)
    
    # Configuration
    means = [50.0, 25.0, 0.5]
    stds = [15.0, 10.0, 0.2]
    
    # Generate data
    df = pipeline.generate_gaussian_data(means, stds)
    
    # Transform
    df_normalized = pipeline.normalize(df)
    
    # Export
    df_normalized.to_csv('synthetic_output.csv', index=False)
    
    print(f"Generated {len(df_normalized)} records")
    print(pipeline.generate_report(df_normalized))
