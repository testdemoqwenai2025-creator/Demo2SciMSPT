WORKFLOW 01

Hello Quantum World

Your first quantum program! Creates a Bell state to demonstrate quantum entanglement and interpret the measurement results. 

# Hello Quantum - Bell State with Results Interpretation
from qiskit import QuantumCircuit, transpile
from qiskit_aer import AerSimulator

# Step 1: Create circuit
qc = QuantumCircuit(2, 2)
qc.h(0)       # Superposition
qc.cx(0, 1)   # Entanglement
qc.measure_all()

# Step 2: Execute
simulator = AerSimulator()
result = simulator.run(transpile(qc, simulator), shots=1024).result()
counts = result.get_counts()

# Step 3: Interpret results
print(f"00 count: {counts.get('00', 0)}")
print(f"11 count: {counts.get('11', 0)}")
print(f"\nEntanglement verified: {abs(counts.get('00',0)-counts.get('11',0))<50}")

WORKFLOW 02

Chemistry Simulation

Calculate H₂ molecule ground state energy using VQE algorithm. A practical template for quantum chemistry research. 

# H2 Molecule Energy Calculation using VQE
from qiskit_nature.second_q.drivers import PySCFDriver
from qiskit_nature.second_q.mappers import JordanWignerMapper
from qiskit_algorithms.minimum_eigensolvers import VQE
from qiskit_algorithms.optimizers import COBYLA
from qiskit.circuit.library import TwoLocal

# Define H2 molecule at equilibrium bond length
driver = PySCFDriver(atom=f"H 0 0 0; H 0 0 0.735")
molecule = driver.run()

# Map to qubit Hamiltonian
mapper = JordanWignerMapper()
qubit_op = mapper.map(molecule.hamiltonian.second_q_op())

# Define variational ansatz (parameterized circuit)
ansatz = TwoLocal(
    rotation_blocks=['ry', 'rz'],
    entanglement_blocks='cx',
    reps=3,
    num_qubits=qubit_op.num_qubits
)

# Run VQE optimization
optimizer = COBYLA(maxiter=200)
vqe = VQE(ansatz, optimizer)
result = vqe.compute_minimum_eigenvalue(qubit_op)

print(f"Ground State Energy: {result.eigenvalue:.6f} Ha")
print(f"Optimal params: {result.optimal_parameters}")

Quantum Machine Learning

VQE-based quantum classifier skeleton. Train quantum circuits for classification tasks using hybrid quantum-classical approach. 

# VQE Classifier Skeleton - Quantum ML Template
import numpy as np
from qiskit import QuantumCircuit
from qiskit_machine_learning.algorithms import VQC
from qiskit_machine_learning.circuit.library import RawFeatureVector
from qiskit_algorithms.optimizers import COBYLA

# Prepare dataset (example: binary classification)
num_features = 2
X_train = np.array([[0, 0], [0, 1], [1, 0], [1, 1]])
y_train = np.array([1, -1, -1, 1])  # XOR labels

# Define quantum feature map
feature_map = RawFeatureMap(feature_dimension=num_features)

# Define ansatz (variational circuit)
ansatz = TwoLocal(
    num_qubits=num_features,
    rotation_blocks=['ry'],
    entanglement_blocks='cx',
    reps=2,
    final_measurement_layer='pz'
)

# Initialize and train classifier
vqc = VQC(
    feature_map=feature_map,
    ansatz=ansatz,
    optimizer=COBYLA(maxiter=100)
)
vqc.fit(X_train, y_train)

# Evaluate
predictions = vqc.predict(X_test)
print(f"Accuracy: {np.mean(predictions == y_test):.2%}")
